> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Transformer `mkrcamp.com` (100% FR) en site bilingue FR + EN avec routing sub-path (FR à la racine, `/en/` avec slugs traduits), back-office FR conservé, et pipeline future-proof pour auto-traduire tout nouveau contenu.

Traduction pilotée par Claude avec sub-agents `general-purpose` en parallèle + glossaire MKR locked + master prompt de traduction. SEO multilingue : hreflang bidirectionnel + sitemap 56 URLs + JSON-LD `inLanguage`. GEO/AI : `llms-en.txt` jumeau + Schema enrichi.

**Tech Stack:** Next.js 16.2.2 (App Router) · TypeScript · next-intl v3+ · framer-motion · CSS vanilla (`globals.css`) · Supabase (@supabase/supabase-js ^2.105) · WeasyPrint (PDF) · Playwright (QA) · GitHub Actions (CI/CD).

```ts
// src/i18n/request.ts
import { getRequestConfig } from 'next-intl/server';
import { hasLocale } from 'next-intl';
import { routing } from './routing';

const NAMESPACES = [
  'common', 'home', 'le-camp', 'programme', 'sessions', 'inscription',
  'familles', 'sur-mesure', 'clubs-groupes', 'destinations', 'temoignages',
  'a-propos', 'contact', 'faq', 'galerie', 'logistique', 'comment-ca-marche',
  'preparer-son-camp', 'guide-caucase', 'merci', 'cgv', 'mentions-legales',
  'politique-de-confidentialite', 'coachs', 'mkr-camp-2026', 'blog',
  'data.sessions', 'data.faq', 'data.testimonials', 'data.registration-types',
  'data.antoine-parcours', 'data.coaches', 'meta',
];

export default getRequestConfig(async ({ requestLocale }) => {
  const requested = await requestLocale;
  const locale = hasLocale(routing.locales, requested) ? requested : routing.defaultLocale;

  const messages: Record<string, unknown> = {};
  for (const ns of NAMESPACES) {
    try {
      messages[ns] = (await import(`../../messages/${locale}/${ns}.json`)).default;
    } catch {
      if (process.env.NODE_ENV !== 'production') {
        console.warn(`[i18n] Missing messages/${locale}/${ns}.json`);
      }
      messages[ns] = {};
    }
  }

  return { locale, messages };
});
```

- [ ] **3.1 Créer `src/middleware.ts`**

```ts
// src/middleware.ts
import createMiddleware from 'next-intl/middleware';
import { NextRequest, NextResponse } from 'next/server';
import { routing } from './i18n/routing';

const intlMiddleware = createMiddleware(routing);

const BOT_PATTERNS = /Googlebot|Bingbot|GPTBot|PerplexityBot|ClaudeBot|YandexBot|DuckDuckBot|facebookexternalhit|Twitterbot|LinkedInBot/i;

export default function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Bloquer /en/admin/* → redirect /admin/*
  if (pathname.startsWith('/en/admin')) {
    const url = request.nextUrl.clone();
    url.pathname = pathname.replace('/en/admin', '/admin');
    return NextResponse.redirect(url);
  }

  // Bots: pas de redirect Accept-Language, on sert la locale demandée par l'URL
  const ua = request.headers.get('user-agent') || '';
  if (BOT_PATTERNS.test(ua)) {
    return intlMiddleware(request);
  }

  return intlMiddleware(request);
}

export const config = {
  matcher: [
    // Match all paths except api, _next, _vercel, static files, admin (kept FR)
    '/((?!api|_next|_vercel|admin|.*\\..*).*)',
  ],
};
```

```ts
// src/lib/i18n-helpers.ts
import { routing, type Locale } from '@/i18n/routing';
import { getPathname } from '@/i18n/navigation';

const SITE_URL = 'https://mkrcamp.com';

export function getAlternateLinks(
  canonicalPath: keyof typeof routing.pathnames,
  currentLocale: Locale,
) {
  const fr = getPathname({ locale: 'fr', href: canonicalPath });
  const en = getPathname({ locale: 'en', href: canonicalPath });

  return {
    canonical: `${SITE_URL}${currentLocale === 'fr' ? fr : en}`,
    languages: {
      fr: `${SITE_URL}${fr}`,
      en: `${SITE_URL}${en}`,
      'x-default': `${SITE_URL}${fr}`,
    },
  };
}

export function localizedMetadata(
  canonicalPath: keyof typeof routing.pathnames,
  locale: Locale,
  title: string,
  description: string,
) {
  const alternates = getAlternateLinks(canonicalPath, locale);
  return {
    title,
    description,
    alternates,
    openGraph: {
      title,
      description,
      url: alternates.canonical,
      locale: locale === 'fr' ? 'fr_FR' : 'en_US',
      siteName: 'MKR Caucasian Camp',
      type: 'website',
    },
    twitter: {
      card: 'summary_large_image',
      title,
      description,
    },
  };
}
```

```tsx
// src/components/LocaleSwitcher.tsx
'use client';

import { useLocale, useTranslations } from 'next-intl';
import { usePathname, useRouter } from '@/i18n/navigation';
import { routing, type Locale } from '@/i18n/routing';

interface Props {
  variant?: 'desktop' | 'mobile';
}

export default function LocaleSwitcher({ variant = 'desktop' }: Props) {
  const currentLocale = useLocale() as Locale;
  const router = useRouter();
  const pathname = usePathname();
  const t = useTranslations('common.locale_switcher');

  function switchTo(target: Locale) {
    if (target === currentLocale) return;
    document.cookie = `NEXT_LOCALE=${target}; max-age=31536000; path=/; SameSite=Lax`;
    router.replace(pathname, { locale: target });
  }

  return (
    <div className={`locale-switcher locale-switcher--${variant}`} aria-label={t('label')}>
      {routing.locales.map((locale) => {
        const isActive = locale === currentLocale;
        return (
          <button
            key={locale}
            type="button"
            className={`locale-switcher-btn ${isActive ? 'is-active' : ''}`}
            aria-current={isActive ? 'page' : undefined}
            aria-label={t(locale === 'fr' ? 'fr' : 'en')}
            lang={locale}
            hrefLang={locale}
            onClick={() => switchTo(locale as Locale)}
            disabled={isActive}
          >
            {t(locale === 'fr' ? 'fr' : 'en')}
          </button>
        );
      })}
    </div>
  );
}
```

```js
// scripts/i18n-check.js
const fs = require('fs');
const path = require('path');

const MESSAGES_DIR = path.join(__dirname, '..', 'messages');
const LOCALES = ['fr', 'en'];

function flattenKeys(obj, prefix = '') {
  const keys = [];
  for (const [k, v] of Object.entries(obj)) {
    const fullKey = prefix ? `${prefix}.${k}` : k;
    if (v && typeof v === 'object' && !Array.isArray(v)) {
      keys.push(...flattenKeys(v, fullKey));
    } else {
      keys.push(fullKey);
    }
  }
  return keys;
}

function loadAllKeys(locale) {
  const localeDir = path.join(MESSAGES_DIR, locale);
  const keys = new Set();

  function walk(dir, namespace = '') {
    for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
      const fullPath = path.join(dir, entry.name);
      if (entry.isDirectory()) {
        walk(fullPath, namespace ? `${namespace}.${entry.name}` : entry.name);
      } else if (entry.name.endsWith('.json')) {
        const ns = namespace
          ? `${namespace}.${entry.name.replace(/\.json$/, '')}`
          : entry.name.replace(/\.json$/, '');
        const content = JSON.parse(fs.readFileSync(fullPath, 'utf8'));
        for (const key of flattenKeys(content)) {
          keys.add(`${ns}::${key}`);
        }
      }
    }
  }

  walk(localeDir);
  return keys;
}

function main() {
  const frKeys = loadAllKeys('fr');
  const enKeys = loadAllKeys('en');

  const missingInEn = [...frKeys].filter((k) => !enKeys.has(k));
  const orphanInEn = [...enKeys].filter((k) => !frKeys.has(k));

  let exitCode = 0;
  if (missingInEn.length > 0) {
    console.error(`Missing EN translations (${missingInEn.length} keys):`);
    for (const k of missingInEn.slice(0, 20)) console.error(`  - ${k}`);
    if (missingInEn.length > 20) console.error(`  ... and ${missingInEn.length - 20} more`);
    exitCode = 1;
  }
  if (orphanInEn.length > 0) {
    console.warn(`Orphan EN keys not in FR (${orphanInEn.length} keys):`);
    for (const k of orphanInEn.slice(0, 10)) console.warn(`  - ${k}`);
  }

  if (exitCode === 0) console.log(`OK: FR and EN have ${frKeys.size} matching keys`);
  process.exit(exitCode);
}

main();
```

```ts
    await page.goto(pair.fr);
    const frAlts = await page.$$eval('link[rel="alternate"]', (links) =>
      links.map((l) => ({ hreflang: l.getAttribute('hreflang'), href: l.getAttribute('href') }))
    );
    expect(frAlts.some((a) => a.hreflang === 'en' && a.href?.endsWith(pair.en))).toBe(true);
    expect(frAlts.some((a) => a.hreflang === 'fr' && a.href?.endsWith(pair.fr))).toBe(true);
    expect(frAlts.some((a) => a.hreflang === 'x-default')).toBe(true);

    // Check EN has alternate to FR
    await page.goto(pair.en);
    const enAlts = await page.$$eval('link[rel="alternate"]', (links) =>
      links.map((l) => ({ hreflang: l.getAttribute('hreflang'), href: l.getAttribute('href') }))
    );
    expect(enAlts.some((a) => a.hreflang === 'fr' && a.href?.endsWith(pair.fr))).toBe(true);
    expect(enAlts.some((a) => a.hreflang === 'en' && a.href?.endsWith(pair.en))).toBe(true);
  });

  test(`canonical per-locale: ${pair.fr} ↔ ${pair.en}`, async ({ page }) => {
    await page.goto(pair.fr);
    const frCanonical = await page.$eval('link[rel="canonical"]', (l) => l.getAttribute('href'));
    expect(frCanonical).toContain(pair.fr);

    await page.goto(pair.en);
    const enCanonical = await page.$eval('link[rel="canonical"]', (l) => l.getAttribute('href'));
    expect(enCanonical).toContain(pair.en);
  });

  test(`JSON-LD inLanguage: ${pair.fr} ↔ ${pair.en}`, async ({ page }) => {
    await page.goto(pair.fr);
    const frScripts = await page.$$eval('script[type="application/ld+json"]', (els) =>
      els.map((e) => JSON.parse(e.textContent || '{}'))
    );
    // Au moins un objet avec inLanguage contenant 'fr'
    const frHasInLang = frScripts.some((s) => {
      const flat = JSON.stringify(s);
      return flat.includes('"inLanguage":"fr"') || flat.includes('"inLanguage":["fr"');
    });
    expect(frHasInLang).toBe(true);

    await page.goto(pair.en);
    const enScripts = await page.$$eval('script[type="application/ld+json"]', (els) =>
      els.map((e) => JSON.parse(e.textContent || '{}'))
    );
    const enHasInLang = enScripts.some((s) => {
      const flat = JSON.stringify(s);
      return flat.includes('"inLanguage":"en"') || flat.includes('"inLanguage":["en"');
    });
    expect(enHasInLang).toBe(true);
  });
}
```
